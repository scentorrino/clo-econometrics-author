\paragraph{Assumptions.}
\begin{enumerate}
\item[\textbf{A1.}] (Sampling) $\{X_i\}_{i=1}^n$ are iid draws from a probability distribution $P$ on $\mathbb{R}$.
\item[\textbf{A2.}] (Integrability) $\mathbb{E}|X| < \infty$, so the population mean $\mu := \mathbb{E}[X]$ is well defined.
\end{enumerate}
